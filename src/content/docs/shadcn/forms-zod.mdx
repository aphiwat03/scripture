## 1. แนวคิด (Concept)

การสร้างแบบฟอร์มในเว็บแอปพลิเคชันระดับองค์กรจำเป็นต้องมี 3 ปัจจัยหลัก:
1. **ประสิทธิภาพ (Performance)**: ไม่กระตุกหรือ Re-render ทั้งหน้าจอเมื่อผู้ใช้พิมพ์แต่ละตัวอักษร
2. **Type Safety & Validation**: ตรวจสอบความถูกต้องของข้อมูลตามกฎเกณฑ์ (Validation Rules) อย่างเข้มงวด ทั้งฝั่ง Client และ Server
3. **Accessibility (a11y)**: เมื่อเกิดข้อผิดพลาด ต้องแจ้งให้ผู้ใช้งานและโปรแกรมอ่านหน้าจอ (Screen Reader) ทราบทันที

shadcn/ui ผสานพลังของ **React Hook Form (RHF)** ร่วมกับ **Zod Schema Validation** ผ่านชุดคอมโพเนนต์โครงสร้าง:
- `<Form>` — ผู้ให้บริการ Context ของ React Hook Form
- `<FormField>` — ตัวเชื่อมประสานระหว่าง RHF Controller กับฟิลด์
- `<FormItem>` — กล่องรวมที่คอยสร้างและส่งต่อ Accessible ID ไปยังลูกหลาน
- `<FormLabel>` — ป้ายชื่อที่ผูกกับช่องกรอกข้อมูลอัตโนมัติ
- `<FormControl>` — ตัวห่อหุ้ม Input, Select หรือ Switch
- `<FormDescription>` — ข้อความอธิบายหรือคำแนะนำเพิ่มเติม
- `<FormMessage>` — ข้อความแสดง Error เมื่อกรอกข้อมูลไม่ถูกต้อง

## 2. หลักการเบื้องหลัง (Under the Hood)

### 1. ทำไมต้องใช้ Uncontrolled Form ใน React Hook Form?
ในวิธีดั้งเดิม (`useState`):
- ทุกครั้งที่พิมพ์ตัวอักษร ฟังก์ชัน `setValues` จะถูกเรียก ส่งผลให้คอมโพเนนต์ทั้งฟอร์มต้องถูก Re-render ใหม่เสมอ
- ใน RHF: ใช้วิธี **Uncontrolled Component ผ่าน Ref** ข้อมูลจะถูกเก็บอยู่ใน DOM หรือหน่วยความจำภายใน และจะสั่ง Re-render เฉพาะฟิลด์ที่สถานะ Error มีการเปลี่ยนแปลงเท่านั้น ทำให้รองรับฟอร์มที่มีฟิลด์นับร้อยได้อย่างลื่นไหล

### 2. มนต์สะกดด้าน Accessibility ใน `FormItem`
เบื้องหลังของ `components/ui/form.tsx`:
- คอมโพเนนต์ `<FormItem>` จะสร้าง Unique ID ขึ้นมาชุดหนึ่งผ่าน `React.useId()`
- `<FormControl>` จะนำ ID นี้ไปใส่ใน `id` ของ Input และผูก `aria-describedby` ชี้ไปยัง ID ของ `<FormDescription>` และ `<FormMessage>`
- เมื่อมี Error เกิดขึ้น `<FormControl>` จะเติม `aria-invalid="true"` ให้อัตโนมัติ ทำให้ Screen Reader ประกาศเสียงทันทีว่า "ข้อมูลช่องนี้ไม่ถูกต้อง: [ข้อความ Error]" โดยที่คุณไม่ต้องเขียนโค้ดผูก ID เองแม้แต่บรรทัดเดียว!

## 3. ตัวอย่างโค้ด (TypeScript / TSX)

### ตัวอย่าง: ฟอร์มสร้าง API Key องค์กร เชื่อมต่อกับ Next.js Server Action

```tsx filename="app/developer/ApiKeyForm.tsx"
"use client";

import * as React from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import * as z from "zod";

import { Button } from "@/components/ui/button";
import {
  Form,
  FormControl,
  FormDescription,
  FormField,
  FormItem,
  FormLabel,
  FormMessage,
} from "@/components/ui/form";
import { Input } from "@/components/ui/input";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { Switch } from "@/components/ui/switch";
import { Loader2, Key, CheckCircle2 } from "lucide-react";

// 1. นิยาม Zod Schema พร้อม Validation Logic
const apiKeySchema = z.object({
  keyName: z
    .string()
    .min(3, { message: "ชื่อคีย์ต้องมีความยาวอย่างน้อย 3 ตัวอักษร" })
    .max(50, { message: "ชื่อคีย์ต้องไม่เกิน 50 ตัวอักษร" }),
  environment: z.enum(["development", "staging", "production"], {
    errorMap: () => ({ message: "กรุณาเลือกสภาพแวดล้อมที่ถูกต้อง" }),
  }),
  ipRestriction: z.boolean().default(false),
  allowedIp: z.string().optional(),
}).refine(
  (data) => {
    // ถ้าเปิดจำกัด IP ต้องระบุ IP Address ให้ถูกต้อง
    if (data.ipRestriction) {
      if (!data.allowedIp || data.allowedIp.trim() === "") return false;
      const ipRegex = /^(?:(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\.){3}(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)$/;
      return ipRegex.test(data.allowedIp);
    }
    return true;
  },
  {
    message: "กรุณาระบุ IPv4 Address ที่ถูกต้อง เช่น 192.168.1.1",
    path: ["allowedIp"],
  }
);

type ApiKeyFormValues = z.infer<typeof apiKeySchema>;

export function ApiKeyForm() {
  const [createdKey, setCreatedKey] = React.useState<string | null>(null);

  // 2. ใช้งาน useForm พร้อม zodResolver
  const form = useForm<ApiKeyFormValues>({
    resolver: zodResolver(apiKeySchema),
    defaultValues: {
      keyName: "",
      environment: "development",
      ipRestriction: false,
      allowedIp: "",
    },
    mode: "onTouched",
  });

  const isSubmitting = form.formState.isSubmitting;
  const isIpRestricted = form.watch("ipRestriction");

  // 3. ฟังก์ชัน Submit ที่ส่งต่อไปยัง Server Action
  async function onSubmit(values: ApiKeyFormValues) {
    // จำลองการเรียก Server Action
    await new Promise((resolve) => setTimeout(resolve, 1500));
    setCreatedKey(`sk_live_${Math.random().toString(36).substring(2, 15)}`);
  }

  return (
    <div className="max-w-xl mx-auto p-6 border rounded-2xl bg-card shadow-xs">
      <div className="flex items-center gap-3 pb-4 mb-6 border-b">
        <div className="p-2.5 rounded-xl bg-primary/10 text-primary">
          <Key className="w-5 h-5" />
        </div>
        <div>
          <h3 className="text-base font-semibold">สร้าง API Key ประจำระบบ</h3>
          <p className="text-xs text-muted-foreground">
            ใช้สำหรับการยืนยันตัวตนผ่าน REST API และ Webhooks
          </p>
        </div>
      </div>

      {createdKey ? (
        <div className="p-4 rounded-xl bg-emerald-500/10 border border-emerald-500/20 text-emerald-950 dark:text-emerald-200 space-y-3">
          <div className="flex items-center gap-2 font-medium text-sm">
            <CheckCircle2 className="w-4 h-4 text-emerald-600" />
            สร้าง API Key เรียบร้อยแล้ว
          </div>
          <p className="text-xs text-muted-foreground">
            กรุณาคัดลอกคีย์นี้ทันที คุณจะไม่สามารถดูคีย์นี้ซ้ำได้อีก
          </p>
          <code className="block p-3 rounded-lg bg-background text-foreground text-xs font-mono border select-all">
            {createdKey}
          </code>
          <Button
            variant="outline"
            size="sm"
            onClick={() => {
              setCreatedKey(null);
              form.reset();
            }}
          >
            สร้างคีย์ใหม่
          </Button>
        </div>
      ) : (
        <Form {...form}>
          <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-6">
            {/* Field: ชื่อคีย์ */}
            <FormField
              control={form.control}
              name="keyName"
              render={({ field }) => (
                <FormItem>
                  <FormLabel>ชื่อคีย์ (Key Name)</FormLabel>
                  <FormControl>
                    <Input placeholder="เช่น Mobile-App-Backend" {...field} />
                  </FormControl>
                  <FormDescription>
                    ตั้งชื่อที่สื่อความหมายเพื่อจดจำแอปพลิเคชันที่นำคีย์นี้ไปใช้งาน
                  </FormDescription>
                  <FormMessage />
                </FormItem>
              )}
            />

            {/* Field: สภาพแวดล้อม */}
            <FormField
              control={form.control}
              name="environment"
              render={({ field }) => (
                <FormItem>
                  <FormLabel>สภาพแวดล้อม (Environment)</FormLabel>
                  <Select onValueChange={field.onChange} defaultValue={field.value}>
                    <FormControl>
                      <SelectTrigger>
                        <SelectValue placeholder="เลือกสภาพแวดล้อม" />
                      </SelectTrigger>
                    </FormControl>
                    <SelectContent>
                      <SelectItem value="development">Development (ทดสอบภายใน)</SelectItem>
                      <SelectItem value="staging">Staging (ระบบจำลองก่อนขึ้นจริง)</SelectItem>
                      <SelectItem value="production">Production (ระบบจริง)</SelectItem>
                    </SelectContent>
                  </Select>
                  <FormMessage />
                </FormItem>
              )}
            />

            {/* Field: จำกัดการเข้าถึงด้วย IP (Switch) */}
            <FormField
              control={form.control}
              name="ipRestriction"
              render={({ field }) => (
                <FormItem className="flex items-center justify-between rounded-xl border p-4">
                  <div className="space-y-0.5">
                    <FormLabel className="text-sm font-semibold">จำกัด IP Address</FormLabel>
                    <FormDescription>
                      อนุญาตให้ส่งคำขอได้เฉพาะ IP ที่ระบุไว้เท่านั้น
                    </FormDescription>
                  </div>
                  <FormControl>
                    <Switch checked={field.value} onCheckedChange={field.onChange} />
                  </FormControl>
                </FormItem>
              )}
            />

            {/* Field: ช่องกรอก IP ที่แสดงตามเงื่อนไข */}
            {isIpRestricted && (
              <FormField
                control={form.control}
                name="allowedIp"
                render={({ field }) => (
                  <FormItem>
                    <FormLabel>IPv4 Address ที่อนุญาต</FormLabel>
                    <FormControl>
                      <Input placeholder="เช่น 203.144.144.1" {...field} />
                    </FormControl>
                    <FormMessage />
                  </FormItem>
                )}
              />
            )}

            <Button type="submit" disabled={isSubmitting} className="w-full gap-2">
              {isSubmitting ? (
                <>
                  <Loader2 className="w-4 h-4 animate-spin" />
                  กำลังสร้างคีย์...
                </>
              ) : (
                "ยืนยันและสร้าง API Key"
              )}
            </Button>
          </form>
        </Form>
      )}
    </div>
  );
}
```

## 4. ผิด vs ถูก

| รูปแบบที่ผิด (Anti-Pattern) | รูปแบบที่ถูกต้อง (Best Practice) |
| :--- | :--- |
| **เขียน `useState` ควบคุมฟอร์ม 15 ฟิลด์**<br/>คอมโพเนนต์ Re-render ทุกครั้งที่กดปุ่มแป้นพิมพ์ ทำให้พิมพ์เร็วๆ แล้วข้อความกระตุก | **ใช้ `useForm` (React Hook Form)**<br/>จัดการ State ภายในแบบ Uncontrolled รวดเร็ว และ Re-render เฉพาะจุดที่ Error |
| **แสดงข้อความ Error ด้วย `<div>` เอง**<br/>```tsx
{errors.email && <div className="text-red-500">{errors.email}</div>}
```<br/>ผลลัพธ์: Screen Reader ไม่รู้ว่าข้อความนี้สัมพันธ์กับช่องกรอกข้อมูลช่องไหน | **ใช้ `<FormMessage />` ภายใน `<FormItem>`**<br/>ระบบจะผูก `aria-describedby` และ `aria-invalid` ให้อัตโนมัติ |
| **ตรวจสอบ Validation ด้วย `if-else` หลายสิบบรรทัดในฟังก์ชัน submit**<br/>ขาด Type safety และไม่สามารถนำโค้ดไปตรวจฝั่ง Server ได้ | **ใช้ Zod Schema พร้อม `zodResolver`**<br/>สร้างกฎชุดเดียว ใช้ตรวจเช็กได้ทั้ง Frontend และ Backend Server Action |

## 5. ควรใช้เมื่อไหร่ / ไม่ควรใช้เมื่อไหร่

### ควรใช้เมื่อ:
- **ฟอร์มที่มีเงื่อนไขตรวจสอบข้อมูล (Validation Logic)**: เช่น เช็ครหัสผ่านตรงกัน, เช็ครูปแบบอีเมล, เช็คเบอร์โทรศัพท์
- **ฟอร์มขนาดกลางถึงใหญ่ในระบบธุรกิจ**: เช่น หน้าตั้งค่าองค์กร, ข้อมูลการจ่ายเงิน, สมัครสมาชิก
- **ต้องการความเข้ากันได้กับ Next.js Server Actions**: ถ่ายโอน Type จาก `z.infer<typeof schema>` ไปยัง Server ได้ทันที

### ไม่ควรใช้เมื่อ:
- **ช่องค้นหาธรรมดาช่องเดียว (Single Search Input)**: สำหรับแถบค้นหาสินค้าบน Header ที่แค่ส่งค่าไปกับ URL Query String ไม่จำเป็นต้องติดตั้ง React Hook Form + Zod ให้หนัก Bundle

## 6. คำถามสัมภาษณ์งาน (Interview Questions)

### Q1: คอมโพเนนต์ `<FormItem>` ใน shadcn/ui ช่วยจัดการเรื่อง Web Accessibility (WCAG 2.1) ให้กับแบบฟอร์มอย่างไรเบื้องหลัง?
**แนวทางตอบ:**
`<FormItem>` ทำหน้าที่เป็น **Accessibility Bridge**:
1. มันสร้าง ID เฉพาะขึ้นมา 1 ชุดโดยใช้ `React.useId()`
2. เมื่อลูกหลานคือ `<FormLabel>` เรนเดอร์ มันจะผูก `htmlFor` กับ ID นี้
3. เมื่อ `<FormControl>` เรนเดอร์ มันจะส่ง `id` นี้ให้กับ Input พร้อมทั้งผูก `aria-describedby` ไปยัง ID ของ `<FormDescription>` และ `<FormMessage>`
4. เมื่อฟิลด์นั้นติดสถานะ Error มันจะใส่ `aria-invalid="true"` ให้อัตโนมัติ ผลลัพธ์คือเครื่อง Screen Reader จะประกาศว่าเกิดข้อผิดพลาดและอ่านข้อความ Error ให้ผู้ใช้ฟังทันทีโดยที่นักพัฒนาไม่ต้องเขียนโค้ดผูก ID เอง

### Q2: จงอธิบายประโยชน์ของการใช้ Zod Schema ร่วมกับ React Hook Form และ Next.js Server Actions ในแง่ของ "Single Source of Truth"?
**แนวทางตอบ:**
1. **แชร์กฎชุดเดียวกัน (Schema Sharing)**: เราสามารถ export ตัวแปร `registerSchema` ตัวเดียวกัน เพื่อใช้ทำ Client-side Validation (แสดง error ทันทีที่ผู้ใช้พิมพ์ผิด) และนำไปใช้ใน Server Action เพื่อทำ Server-side Validation ซ้ำอีกครั้ง
2. **End-to-End Type Safety**: การใช้ `z.infer<typeof registerSchema>` ช่วยสร้าง TypeScript Type ขึ้นมาโดยตรง เมื่อมีการแก้ไขกฎใน Zod (เช่น เพิ่มความยาวรหัสผ่าน) ทั้งหน้า UI และฟังก์ชันในเซิร์ฟเวอร์จะอัปเดต Type อัตโนมัติ หากมีจุดที่เขียนผิด TypeScript จะแจ้งเตือนทันทีช่วงคอมไพล์ ป้องกันบั๊กได้อย่างสมบูรณ์
